
import boto3

def get_secret(secret_name: str) -> str:
    region_name = "us-east-1"

    # Create a Secrets Manager client
    client = boto3.client(service_name="secretsmanager", region_name=region_name)

    try:
        get_secret_value_response = client.get_secret_value(SecretId=secret_name)
        secret = json.loads(get_secret_value_response["SecretString"])
        if secret is None:
            raise Exception
        return secret
    except ClientError as e:
        print(f"Unable to retrieve secret: {e}")
        # For a list of exceptions thrown, see
        # https://docs.aws.amazon.com/secretsmanager/latest/apireference/API_GetSecretValue.html
        raise e
    except Exception as e:
        print(f"Unknown exception: Unable to retrieve secret: {e}")
        raise e

TOKEN_FILE=os.path.join(os.path.expanduser('~'), ".insilicoclitoken")

def write_cache(cache: SerializableTokenCache, file: str):
    if cache.has_state_changed:
        with open(file, "w") as f:
            f.write(cache.serialize())

def post_data(bearer_token, url, data):
    headers = {
        'accept': 'application/json',
        'Content-Type': 'application/json',
        'Authorization': f'Bearer {bearer_token}'
    }
    response = requests.post(url, headers=headers, data=json.dumps(data))
    return response.json()
